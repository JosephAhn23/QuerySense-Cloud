CLI equivalent (for scripting) 
$ querysense workbook init --sql "SELECT * FROM orders WHERE user_id = \$1" --name my_opt
$ querysense workbook add-params --name my_opt --set "user_id=1,status=shipped"
$ querysense workbook add-params --name my_opt --set "user_id=50000,status=pending"
$ querysense workbook baseline --name my_opt --dsn $DATABASE_URL
$ querysense workbook add-variant --name my_opt --variant composite_idx \
    --index-sql "CREATE INDEX idx_orders_covering ON orders(user_id, status, created_at DESC)"
$ querysense workbook test --name my_opt --dsn $DATABASE_URL
$ querysense workbook compare --name my_opt
$ querysense workbook apply --name my_opt --output migration.sql
